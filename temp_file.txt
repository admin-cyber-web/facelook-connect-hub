const totalLikesFrom = (row: any, fallback = 0) => {
  const raw = row?.total_likes ?? row?.likes_count ?? row?.like_count ?? fallback;
  const parsed = Number(raw);
  return Number.isFinite(parsed) && parsed >= 0 ? parsed : fallback;
};

function normalizePostLikeCount<T extends Record<string, any>>(
  post: T,
): T & { likes_count: number; content: string } {
  let metadata = post.metadata;
  if (typeof metadata === "string") {
    try {
      metadata = JSON.parse(metadata);
    } catch {
      metadata = null;
    }
  }
  return {
    ...post,
    content: getPostCaption({ ...post, metadata }),
    likes_count: totalLikesFrom(post),
    surpriseMsg: post.surpriseMsg ?? metadata?.surpriseMsg ?? null,
  };
}

type SurpriseMessage = {
  targetUserId: string;
  targetUserName: string;
  customMessage: string;
  gifUrl: string;
  isSeen: boolean;
};

function getSurpriseMessage(post: any): SurpriseMessage | null {
  let raw = post?.surpriseMsg ?? post?.metadata?.surpriseMsg;
  if (typeof raw === "string") {
    try {
      raw = JSON.parse(raw);
    } catch {
      return null;
    }
  }
  if (!raw || typeof raw !== "object" || !raw.targetUserId) return null;
  return {
    targetUserId: String(raw.targetUserId),
    targetUserName: String(raw.targetUserName || "friend"),
    customMessage: String(raw.customMessage || ""),
    gifUrl: String(raw.gifUrl || ""),
    isSeen: raw.isSeen === true,
  };
}

// â”€â”€ PostViewTracker â€” fires onView once when post scrolls into view â”€â”€â”€â”€â”€â”€â”€â”€â”€
const VibeVisibilityContext = createContext(false);
const FeedVideoPreloadContext = createContext<{
  activeIndex: number | null;
  setActiveIndex: (
    index: number | null | ((current: number | null) => number | null),
  ) => void;
} | null>(null);

const PostViewTracker = memo(({
  postId,
  onView,
  vibeAudioUrl,
  vibeAudioLoop = true,
  children,
}: {
  postId: string;
  onView: (id: string) => void;
  vibeAudioUrl?: string | null;
  vibeAudioLoop?: boolean;
  children: React.ReactNode;
}) => {
  const ref = useRef<HTMLDivElement>(null);
  const fired = useRef(false);
  const [isVisible, setIsVisible] = useState(false);
  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const obs = new IntersectionObserver(
      ([entry]) => {
        const visible = entry.isIntersecting && entry.intersectionRatio >= 0.1;
        setIsVisible(visible);
        setVisibleVibeAudio(
          postId,
          vibeAudioUrl ?? null,
          visible,
          vibeAudioLoop,
          entry.intersectionRatio,
        );
        if (entry.isIntersecting && entry.intersectionRatio >= 0.5 && !fired.current) {
          fired.current = true;
          onView(postId);
        }
      },
      { threshold: [0.1, 0.5] },
    );
    obs.observe(el);
    return () => {
      obs.disconnect();
      setVisibleVibeAudio(postId, vibeAudioUrl ?? null, false, vibeAudioLoop);
    };
  }, [postId, onView, vibeAudioUrl, vibeAudioLoop]);
  return (
    <VibeVisibilityContext.Provider value={isVisible}>
      <div ref={ref} className={isVisible ? "post-vibe-in-view" : undefined} style={{ touchAction: "pan-y" }}>
        {children}
      </div>
    </VibeVisibilityContext.Provider>
  );
});

// â”€â”€ Inline video â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
const FeedVideo = memo(({ src, poster, isCreator, feedIndex }: { src: string; poster?: string | null; isCreator: boolean; feedIndex?: number }) => {
  const ref = useRef<HTMLVideoElement>(null);
  const isVisibleRef = useRef(false);
  const [isVisible, setIsVisible] = useState(false);
  const [isMuted, setIsMuted] = useState(true);
  const [videoError, setVideoError] = useState<string | null>(null);
  const preloadContext = useContext(FeedVideoPreloadContext);
  const setActiveFeedVideoIndex = preloadContext?.setActiveIndex;
  const bunnyVideoId = getBunnyVideoId(src);
  const inPreloadWindow =
    feedIndex !== undefined &&
    preloadContext?.activeIndex !== null &&
    preloadContext?.activeIndex !== undefined &&
    feedIndex >= preloadContext.activeIndex &&
    feedIndex <= preloadContext.activeIndex + 2;
  const bunnyVideoStatus = useBunnyVideoStatus(
    bunnyVideoId,
    isCreator && (isVisible || inPreloadWindow),
  );
  const hlsPlayback = useHlsVideo(
    ref,
    src,
    (isVisible || inPreloadWindow) &&
      (!isCreator || bunnyVideoStatus.phase === "ready"),
    bunnyVideoStatus.phase === "ready" ? "ready" : "pending",
    isVisible,
    !isVisible,
  );

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    const obs = new IntersectionObserver(
      ([entry]) => {
        const visible = entry.isIntersecting && entry.intersectionRatio >= 0.35;
        isVisibleRef.current = visible;
        setIsVisible(visible);
        if (visible) {
          if (feedIndex !== undefined) setActiveFeedVideoIndex?.(feedIndex);
          el.muted = false;
          el.volume = 1;
          setIsMuted(false);
          playVideoWithMutedFallback(el, () => {
            if (!isVisibleRef.current) return false;
            setIsMuted(true);
          });
        } else {
          el.pause();
          el.muted = true;
          setIsMuted(true);
          if (feedIndex !== undefined) {
            setActiveFeedVideoIndex?.((current) =>
              current === feedIndex ? null : current,
            );
          }
        }
      },
      { threshold: [0, 0.35, 0.5] },
    );
    obs.observe(el);
    return () => {
      obs.disconnect();
      el.pause();
    };
  }, [feedIndex, setActiveFeedVideoIndex, src]);

  useEffect(() => {
    if (!isVisible || (isCreator && bunnyVideoStatus.phase !== "ready")) return;
    const video = ref.current;
    if (!video) return;
    setVideoError(null);
    if (!hlsPlayback.isHls) {
      video.load();
      playVideoWithMutedFallback(video, () => {
        if (!isVisibleRef.current) return false;
        setIsMuted(true);
      });
    }
  }, [bunnyVideoStatus.phase, hlsPlayback.isHls, isCreator, isVisible, src]);

  return (
    <div
      className="relative mx-auto my-3 aspect-[9/16] w-[min(100%,42.1875vh)] max-h-[75vh] overflow-hidden rounded-2xl bg-black md:w-[min(100%,45vh)] md:max-h-[80vh]"
      style={{ touchAction: "pan-y" }}
    >
      <video
        ref={ref}
        src={hlsPlayback.isHls ? undefined : src}
        poster={getBunnyVideoThumbnailUrl(src, poster)}
        loop
        muted={isMuted}
        playsInline
        crossOrigin="anonymous"
        autoPlay={isVisible}
        className="w-full h-full rounded-2xl object-cover"
        style={{ touchAction: "pan-y" }}
        preload={isVisible || inPreloadWindow ? "auto" : "metadata"}
        {...{
          "webkit-playsinline": "true",
          "x5-playsinline": "true",
        }}
        onLoadStart={() => setVideoError(null)}
        onError={(event) => {
          if (isCreator && bunnyVideoId && bunnyVideoStatus.phase !== "ready") return;
          setVideoError("This video couldn't be played. Try again later.");
          console.warn("[FameFeed] Video playback failed:", event.currentTarget.error, src);
        }}
      />
      {isVisible && isMuted && (
        <button
          type="button"
          className="absolute bottom-3 right-3 z-20 flex cursor-pointer items-center gap-2 rounded-full border border-white/25 bg-black/65 px-3 py-2 text-xs font-bold text-white shadow-lg backdrop-blur transition hover:bg-black/80"
          aria-label="Enable video sound"
          onClick={(event) => {
            event.stopPropagation();
            const video = ref.current;
            if (!video) return;
            video.muted = false;
            video.volume = 1;
            void requestVideoPlay(video).then(
              () => setIsMuted(false),
              () => {
                video.muted = true;
                setIsMuted(true);
                playVideoWithMutedFallback(video, () => setIsMuted(true));
              },
            );
          }}
        >
          <VolumeX size={14} aria-hidden="true" />
          Tap for sound
        </button>
      )}
      {isCreator && bunnyVideoId && (
        <>
          <BunnyCreatorProcessingOverlay state={bunnyVideoStatus} />
          {bunnyVideoStatus.phase === "failed" && (
            <BunnyVideoStatusOverlay state={bunnyVideoStatus} onRetry={bunnyVideoStatus.retry} />
          )}
        </>
      )}
      {(videoError || (!isCreator || bunnyVideoStatus.phase === "ready"
        ? hlsPlayback.error
        : null)) && (
        <div className="absolute inset-0 z-[25] flex flex-col items-center justify-center gap-2 bg-[#090b14]/95 p-4 text-center">
          <p className="text-sm font-bold text-white">
            {videoError || hlsPlayback.error}
          </p>
          <button
            type="button"
            className="rounded-full border border-white/20 bg-white/10 px-4 py-2 text-xs font-bold text-white"
            onClick={() => {
              const video = ref.current;
              if (!video) return;
              setVideoError(null);
              if (hlsPlayback.isHls) {
                hlsPlayback.retry();
              } else {
                video.load();
                void video.play().catch(() => {});
              }
            }}
          >
            Retry
          </button>
        </div>
      )}
    </div>
  );
});

const getPostImageUrls = (post: any): string[] => {
  let rawUrls = post?.image_urls ?? post?.metadata?.image_urls;
  if (typeof rawUrls === "string") {
    try {
      rawUrls = JSON.parse(rawUrls);
    } catch {
      rawUrls = rawUrls.includes(",") ? rawUrls.split(",") : [rawUrls];
    }
  }
  if (Array.isArray(rawUrls)) {
    const urls = rawUrls.filter((url): url is string => typeof url === "string" && url.trim().length > 0);
    if (urls.length) return urls.slice(0, 5);
  }

  const legacyUrl = post?.media_url || post?.image_url || post?.cover_url || null;
  if (!legacyUrl || post?.type === "video") return [];
  const isVideo =
    post?.metadata?.is_youtube ||
    legacyUrl.includes("youtube.com") ||
    legacyUrl.includes("youtu.be") ||
    /\.(mp4|webm|ogg|mov|m4v)/i.test(legacyUrl.split("?")[0]) ||
    legacyUrl.includes("rapidcdn.app");
  return isVideo ? [] : [legacyUrl];
};

const LatestSurveysWidget = ({
  currentUserId,
  onNavigateToSurveys,
}: {
  currentUserId: string;
  onNavigateToSurveys?: () => void;
}) => {
  const [surveys, setSurveys]         = useState<any[]>([]);
  const [loaded, setLoaded]           = useState(false);
  const [voteCounts, setVoteCounts]   = useState<Record<string, Record<string, number>>>({});

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const { data } = await supabase
        .from("surveys")
        .select("id, question, image_url, media_url, user_id, created_at, survey_options(id, text)")
        .order("created_at", { ascending: false })
        .limit(3);
      if (cancelled) return;
      setSurveys(data || []);
      setLoaded(true);
      if (data && data.length > 0) {
        const ids = data.map((s: any) => s.id);
        const { data: voteCounts } = await supabase.rpc("get_survey_vote_counts", {
          p_survey_ids: ids,
        });
        if (!cancelled && voteCounts) {
          const map: Record<string, Record<string, number>> = {};
          voteCounts.forEach((v: any) => {
            if (!map[v.survey_id]) map[v.survey_id] = {};
            map[v.survey_id][v.option_id] = Number(v.vote_count) || 0;
          });
          setVoteCounts(map);
        }
      }
    })();
    return () => { cancelled = true; };
  }, []);

  if (!loaded || surveys.length === 0) return null;

  return (
    <div className="pt-3 pb-2 border-b border-white/5" style={{ background: "#0F172A" }}>
      {/* Section header */}
      <div className="flex items-center justify-between px-4 mb-3">
        <p className="text-[13px] font-black text-white/70 tracking-wide">ðŸ—³ï¸ Latest Surveys</p>
        {onNavigateToSurveys && (
          <button
            onClick={onNavigateToSurveys}
            className="text-[11px] font-bold text-indigo-400/70 hover:text-indigo-400 transition-colors">
            See all â†’
          </button>
        )}
      </div>

      {/* Cards â€” horizontal scroll on mobile, single column on md+ */}
      <div className="flex gap-3 overflow-x-auto px-4 pb-2 no-scrollbar sm:flex-col sm:overflow-x-visible sm:gap-2">
        {surveys.map((survey, idx) => (
          <motion.div
            key={survey.id}
            initial={{ opacity: 0, y: 6 }}
            animate={{ opacity: 1, y: 0 }}
            transition={{ delay: idx * 0.07 }}
            onClick={onNavigateToSurveys}
            className="flex-shrink-0 w-56 sm:w-full rounded-2xl overflow-hidden cursor-pointer active:scale-[0.98] transition-transform"

const SUGG_GRADS = [
  "#6366f1",
  "#ec4899",
  "#f59e0b",
  "#10b981",
  "#3b82f6",
  "#8b5cf6",
];

function gradForSugg(id: string) {
  let h = 0;
  for (let i = 0; i < id.length; i++) h = ((h << 5) + h) ^ id.charCodeAt(i);
  return SUGG_GRADS[Math.abs(h) % SUGG_GRADS.length];
}

// Seeded Fisher-Yates shuffle â€” deterministic per seed, stable within a render
function seededShuffle<T>(arr: T[], seed: number): T[] {
  if (arr.length === 0) return [];
  const a = [...arr];
  let s = (seed * 1664525 + 1013904223) & 0x7fffffff;
  for (let i = a.length - 1; i > 0; i--) {
    s = (s * 1664525 + 1013904223) & 0x7fffffff;
    const j = s % (i + 1);
    [a[i], a[j]] = [a[j], a[i]];
  }
  return a;
}

let activeFameFeedVideo: HTMLVideoElement | null = null;
let activeFameFeedAudio: HTMLAudioElement | null = null;

type FameFeedReelSettingsRef = {
  current: {
    muted: boolean;
    hasBackgroundAudio: boolean;
    setAudioNeedsGesture: (blocked: boolean) => void;
    onActivate: () => void;
  };
};

type FameFeedReelCandidate = {
  ratio: number;
  centerDistance: number;
  audio: HTMLAudioElement | null;
  settings: FameFeedReelSettingsRef;
};

const visibleFameFeedReels = new Map<HTMLVideoElement, FameFeedReelCandidate>();
let lastCenteredFameFeedVideo: HTMLVideoElement | null = null;

const isConnectedMediaElement = (element: HTMLMediaElement | null): element is HTMLMediaElement =>
  Boolean(element && element.isConnected && typeof element.pause === "function");

const pauseAndMuteFameFeedReel = (video: HTMLVideoElement | null, audio: HTMLAudioElement | null) => {
  if (isConnectedMediaElement(video)) {
    try {
      video.pause();
      video.muted = true;
    } catch {}
  }
  if (isConnectedMediaElement(audio)) {
    try {
      audio.pause();
      audio.muted = true;
    } catch {}
  }
};

const syncCenteredFameFeedReel = () => {
  [...visibleFameFeedReels.entries()].forEach(([video, reel]) => {
    if (!isConnectedMediaElement(video)) {
      visibleFameFeedReels.delete(video);
      if (isConnectedMediaElement(reel.audio)) {
        try {
          reel.audio.pause();
          reel.audio.muted = true;
        } catch {}
      }
      return;
    }
    const rect = video.getBoundingClientRect();
    const visibleHeight = Math.max(0, Math.min(rect.bottom, window.innerHeight) - Math.max(rect.top, 0));
    const ratio = rect.height > 0 ? visibleHeight / rect.height : 0;
    const centerDistance = Math.abs(rect.top + rect.height / 2 - window.innerHeight / 2);
    if (ratio <= 0) {
      visibleFameFeedReels.delete(video);
    } else {
      visibleFameFeedReels.set(video, { ...reel, ratio, centerDistance });
    }
  });
  const candidate = [...visibleFameFeedReels.entries()]
    .filter(([video, reel]) => isConnectedMediaElement(video) && reel.ratio > 0)
    .sort(([, first], [, second]) => first.centerDistance - second.centerDistance || second.ratio - first.ratio)[0];
  const activeVideo = candidate?.[0] ?? null;
  const activeReel = candidate?.[1] ?? null;

  visibleFameFeedReels.forEach((reel, video) => {
    if (video === activeVideo) return;
    if (isConnectedMediaElement(video)) {
      try {
        video.pause();
        video.muted = true;
      } catch {}
    }
    if (isConnectedMediaElement(reel.audio)) {
      try {
        reel.audio.pause();
        reel.audio.muted = true;
      } catch {}
    }
  });

  activeFameFeedVideo = activeVideo;
  activeFameFeedAudio = activeReel?.audio ?? null;
  if (lastCenteredFameFeedVideo !== activeVideo) {
    lastCenteredFameFeedVideo = activeVideo;
    activeReel?.settings.current.onActivate();
  }
  if (!isConnectedMediaElement(activeVideo) || !activeReel) return;

  const settings = activeReel.settings.current;
  try {
    activeVideo.muted = settings.hasBackgroundAudio || settings.muted;
    activeVideo.volume = 1;
    if (activeVideo.paused) void activeVideo.play().catch(() => {});
  } catch {}

  if (!isConnectedMediaElement(activeReel.audio) || !settings.hasBackgroundAudio) return;
  try {
    activeReel.audio.muted = settings.muted;
    activeReel.audio.volume = 1;
  } catch {
    return;
  }
  if (settings.muted) {
    try { activeReel.audio.pause(); } catch {}
    return;
  }
  try {
    if (Math.abs(activeReel.audio.currentTime - activeVideo.currentTime) > 0.18) {
      activeReel.audio.currentTime = activeVideo.currentTime;
    }
    if (activeReel.audio.paused) {
      void activeReel.audio.play()
        .then(() => settings.setAudioNeedsGesture(false))
        .catch(() => settings.setAudioNeedsGesture(true));
    }
  } catch {
    settings.setAudioNeedsGesture(true);
  }
};

let fameReelSyncFrame = 0;
const scheduleCenteredFameFeedReelSync = () => {
  if (document.hidden || visibleFameFeedReels.size === 0 || fameReelSyncFrame) return;
  fameReelSyncFrame = window.requestAnimationFrame(() => {
    fameReelSyncFrame = 0;
    if (document.hidden) return;
    syncCenteredFameFeedReel();
  });
};
const cancelCenteredFameFeedReelSync = () => {
  if (!fameReelSyncFrame) return;
  window.cancelAnimationFrame(fameReelSyncFrame);
  fameReelSyncFrame = 0;
};
const handleFameFeedVisibilityChange = () => {
  if (document.hidden) {
    cancelCenteredFameFeedReelSync();
    return;
  }
  scheduleCenteredFameFeedReelSync();
};

const SingleReelBlock = ({

  feedIndex?: number;

  onOpenComments: () => void;
  onEdit: () => void;
  onDelete: () => void;
  onReport: () => void;
}) => {
  const onlineUserIds = useOnlineUsers();

  const viewerPreviewLoadedRef = useRef(false);
  const [likedByMe, setLikedByMe] = useState(false);
  const [likeCount, setLikeCount] = useState(() => totalLikesFrom(post));
  const [likePending, setLikePending] = useState(false);
  const likeMutationRef = useRef<Promise<void> | null>(null);
  const reelSettings = getReelSettings(post);
  const reelVideoUrl = reelSettings.videoUrl || post.media_url;
  const preloadContext = useContext(FeedVideoPreloadContext);
  const setActiveFeedVideoIndex = preloadContext?.setActiveIndex;
  const inPreloadWindow =
    feedIndex !== undefined &&
    preloadContext?.activeIndex !== null &&
    preloadContext?.activeIndex !== undefined &&
    feedIndex >= preloadContext.activeIndex - 3 &&
    feedIndex <= preloadContext.activeIndex + 5;
  const [isHlsActive, setIsHlsActive] = useState(false);
  const isOwner = Boolean(currentUserId && post.author_id === currentUserId);
  const bunnyVideoId = getBunnyVideoId(reelVideoUrl, post?.metadata?.bunny_video_id);
  const bunnyVideoStatus = useBunnyVideoStatus(bunnyVideoId, isOwner && isHlsActive);
  const hlsPlayback = useHlsVideo(
    ref,
    reelVideoUrl,
    (isHlsActive || inPreloadWindow) &&
      (!isOwner || bunnyVideoStatus.phase === "ready"),
    bunnyVideoStatus.phase === "ready" ? "ready" : "pending",
    false,
    !isHlsActive,
  );
  const hasBackgroundAudio = Boolean(reelSettings.audioUrl);
  const toggleReelMenu = (event: React.SyntheticEvent<HTMLButtonElement>) => {
    event.stopPropagation();
    if (event.cancelable) event.preventDefault();
    if (showReelMenu) {
      setShowReelMenu(false);
      return;
    }

    const rect = event.currentTarget.getBoundingClientRect();
    const menuWidth = Math.min(176, Math.max(160, window.innerWidth - 24));
    const maxRight = Math.max(12, window.innerWidth - menuWidth - 12);
    const right = Math.max(
      12,
      Math.min(maxRight, window.innerWidth - rect.right),
    );
    const estimatedMenuHeight = isOwner ? 120 : 64;
    const top =
      rect.bottom + estimatedMenuHeight + 8 <= window.innerHeight
        ? rect.bottom + 8
        : Math.max(12, rect.top - estimatedMenuHeight - 8);

    setReelMenuPosition({ top, right });
    setShowReelMenu(true);
  };
  const toggleReelMenuFromKeyboard = (
    event: React.KeyboardEvent<HTMLButtonElement>,
  ) => {
    if (event.repeat || (event.key !== "Enter" && event.key !== " ")) return;
    toggleReelMenu(event);
  };
  const runReelMenuAction = (
    event: React.SyntheticEvent<HTMLButtonElement>,
    actionType: string,
    action: () => void,
  ) => {
    event.stopPropagation();
    if (event.cancelable) event.preventDefault();
    console.log("ACTION TRIGGERED:", actionType, String(post.id));
    try {
      action();
    } finally {
      setShowReelMenu(false);
    }
  };
  const runReelMenuActionFromKeyboard = (
    event: React.KeyboardEvent<HTMLButtonElement>,
    actionType: string,
    action: () => void,
  ) => {
    if (event.repeat || (event.key !== "Enter" && event.key !== " ")) return;
    runReelMenuAction(event, actionType, action);
  };
  useEffect(() => {
    if (!showReelMenu) return;
    const closeOnOutsidePointerDown = (event: PointerEvent) => {
      const target = event.target;
      if (!(target instanceof Element)) return;
      if (
        target.closest("[data-fame-reel-menu]") ||
        target.closest("[data-fame-reel-menu-trigger]")
      ) {
        return;
      }
      setShowReelMenu(false);
    };
    const closeOnEscape = (event: KeyboardEvent) => {
      if (event.key === "Escape") setShowReelMenu(false);
    };

    document.addEventListener("pointerdown", closeOnOutsidePointerDown, true);
    document.addEventListener("keydown", closeOnEscape);
    return () => {
      document.removeEventListener(
        "pointerdown",
        closeOnOutsidePointerDown,
        true,
      );
      document.removeEventListener("keydown", closeOnEscape);
    };
  }, [showReelMenu]);
  // Warm attached audio without duplicating video downloads; the video element
  // and HLS hook manage their own small, viewport-based preload window.
  useEffect(() => {
    if (reelSettings.audioUrl) prefetchReelMedia(reelSettings.audioUrl);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [reelVideoUrl, reelSettings.audioUrl]);
  const loadViewerPreviews = useCallback(async () => {
    if (viewerPreviewLoadedRef.current) return;
    viewerPreviewLoadedRef.current = true;
    try {
      const { data: viewRows, error: viewError } = await supabase
        .from("post_views")
        .select("user_id")
        .eq("post_id", post.id)
        .not("user_id", "is", null)
        .order("viewed_at", { ascending: false })
        .limit(6);
      if (viewError || !viewRows?.length) return;
      const viewerIds = [...new Set(viewRows.map((row: any) => row.user_id).filter((id: string) => id && id !== currentUserId))].slice(0, 3);
      if (!viewerIds.length) return;
      const { data: profiles, error: profilesError } = await supabase
        .from("profiles")
        .select("id, full_name, avatar_url")
        .in("id", viewerIds);
      if (profilesError || !profiles) return;
      const profilesById = new Map(profiles.map((profile: any) => [profile.id, profile]));
      setViewerPreviews(viewerIds.map((id) => profilesById.get(id)).filter(Boolean) as Array<{ id: string; full_name: string; avatar_url: string | null }>);
    } catch (error) {
      console.warn("[FameFeed] reel audience preview failed:", error);
    }
  }, [currentUserId, post.id]);
  const playbackSettingsRef = useRef({ muted, hasBackgroundAudio, setAudioNeedsGesture, onActivate: loadViewerPreviews });
  playbackSettingsRef.current = { muted, hasBackgroundAudio, setAudioNeedsGesture, onActivate: loadViewerPreviews };

  const refreshLikeState = useCallback(async () => {
    const countRequest = supabase
      .from("likes")
      .select("id", { count: "exact", head: true })
      .eq("post_id", post.id);
    const mineRequest = currentUserId
      ? supabase
          .from("likes")
          .select("id")
          .eq("post_id", post.id)
          .eq("user_id", currentUserId)
          .limit(1)
      : Promise.resolve({ data: [], error: null });

    const [countResult, mineResult] = await Promise.all([
      countRequest,
      mineRequest,
    ]);

    if (countResult.error || mineResult.error) {
      console.warn(
        "[FameFeed][reel-like] state refresh failed:",
        countResult.error?.message || mineResult.error?.message,
      );
      return;
    }

    setLikeCount(countResult.count ?? 0);
    setLikedByMe(Boolean(mineResult.data?.length));
  }, [currentUserId, post.id]);

  useEffect(() => {
    let cancelled = false;
    void (async () => {
      const countRequest = supabase
        .from("likes")
        .select("id", { count: "exact", head: true })
        .eq("post_id", post.id);
      const mineRequest = currentUserId
        ? supabase
            .from("likes")
            .select("id")
            .eq("post_id", post.id)
            .eq("user_id", currentUserId)
            .limit(1)
        : Promise.resolve({ data: [], error: null });
      const [countResult, mineResult] = await Promise.all([
        countRequest,
        mineRequest,
      ]);
      if (cancelled) return;
      if (countResult.error || mineResult.error) {
        console.warn(
          "[FameFeed][reel-like] initial state load failed:",
          countResult.error?.message || mineResult.error?.message,
        );
        return;
      }
      setLikeCount(countResult.count ?? 0);
      setLikedByMe(Boolean(mineResult.data?.length));
    })();
    return () => {
      cancelled = true;
    };
  }, [currentUserId, post.id]);

  // Keep a mounted reel in sync if its parent feed receives a refreshed
  // aggregate row while the reel is on screen.
  useEffect(() => {
    setLikeCount(totalLikesFrom(post));
  }, [post.id, post.likes_count, post.total_likes]);

  const toggleReelLike = useCallback(async () => {
    if (!currentUserId) {
      toast.error("Please login to like");
      return;
    }
    if (likeMutationRef.current) return;

    const nextLiked = !likedByMe;
    setLikePending(true);
    setLikedByMe(nextLiked);
    setLikeCount((count) => Math.max(count + (nextLiked ? 1 : -1), 0));

    const mutation = (async () => {
      const result = nextLiked
        ? await supabase
            .from("likes")
            .upsert(
              { post_id: post.id, user_id: currentUserId },
              { onConflict: "post_id,user_id", ignoreDuplicates: true },
            )
        : await supabase
            .from("likes")
            .delete()
            .eq("post_id", post.id)
            .eq("user_id", currentUserId);

      if (result.error) {
        toast.error(nextLiked ? "Like nahi ho saka." : "Like nahi hata.");
      }

      // Always reconcile from likes rows. This handles duplicate taps,
      // concurrent sessions, and an idempotent upsert/delete response.
      await refreshLikeState();
    })();

    likeMutationRef.current = mutation;
    try {
      await mutation;
    } finally {
      if (likeMutationRef.current === mutation) {
        likeMutationRef.current = null;
        setLikePending(false);
      }
    }
  }, [currentUserId, likedByMe, post.id, refreshLikeState]);

  useEffect(() => {
    const el = ref.current;
    if (!el) return;
    el.playbackRate = reelSettings.playbackRate;
    el.style.filter = reelSettings.cssFilter;
    const retryPlayback = () => {
      const settings = playbackSettingsRef.current;
      if (!isConnectedMediaElement(ref.current) || activeFameFeedVideo !== ref.current || document.visibilityState === "hidden") return;
      const activeVideo = ref.current;
      try {
        activeVideo.muted = settings.hasBackgroundAudio || settings.muted;
        activeVideo.volume = 1;
        if (activeVideo.paused) void activeVideo.play().catch(() => {});
      } catch {}
      const audio = audioRef.current;
      if (isConnectedMediaElement(audio) && settings.hasBackgroundAudio) {
        try {
          audio.muted = settings.muted;
          audio.volume = 1;
          if (!settings.muted && audio.paused) {
            void audio.play().then(() => settings.setAudioNeedsGesture(false)).catch(() => settings.setAudioNeedsGesture(true));
          }
        } catch {
          settings.setAudioNeedsGesture(true);
        }
      }
    };
    const gestureEvents = ["pointerdown", "touchstart", "click", "scroll", "wheel"] as const;
    gestureEvents.forEach((eventName) => document.addEventListener(eventName, retryPlayback, { capture: true, passive: true }));
    const obs = new IntersectionObserver(
      (entries) => {
        const entry = entries[0];
        const video = ref.current;
        if (!entry || !isConnectedMediaElement(video)) {
          setIsHlsActive(false);
          if (video) {
            visibleFameFeedReels.delete(video);
            pauseAndMuteFameFeedReel(video, audioRef.current);
          }
          syncCenteredFameFeedReel();
          return;
        }
        const centerDistance = Math.abs(
          entry.boundingClientRect.top + entry.boundingClientRect.height / 2 - window.innerHeight / 2,
        );
        if (entry.isIntersecting && entry.intersectionRatio > 0) {
          setIsHlsActive(entry.intersectionRatio >= 0.35);
          if (entry.intersectionRatio >= 0.35 && feedIndex !== undefined) {
            setActiveFeedVideoIndex?.(feedIndex);
          }
          visibleFameFeedReels.set(video, {
            ratio: entry.intersectionRatio,
            centerDistance,
            audio: audioRef.current,
            settings: playbackSettingsRef,
          });
        } else {
          setIsHlsActive(false);
          if (feedIndex !== undefined) {
            setActiveFeedVideoIndex?.((current) =>
              current === feedIndex ? null : current,
            );
          }
          visibleFameFeedReels.delete(video);
          pauseAndMuteFameFeedReel(video, audioRef.current);
        }
        syncCenteredFameFeedReel();
      },
      { threshold: [0, 0.35, 0.5, 0.75, 1] },
    );
    try {
      if (el.isConnected) obs.observe(el);
    } catch {
      visibleFameFeedReels.delete(el);
    }
    return () => {
      obs.disconnect();
      gestureEvents.forEach((eventName) => document.removeEventListener(eventName, retryPlayback, true));
      visibleFameFeedReels.delete(el);
      pauseAndMuteFameFeedReel(el, audioRef.current);
      syncCenteredFameFeedReel();
    };
  }, [feedIndex, setActiveFeedVideoIndex, reelSettings.cssFilter, reelSettings.playbackRate, reelVideoUrl]);

  useEffect(() => {
    const video = ref.current;
    const audio = audioRef.current;
    if (!isConnectedMediaElement(video) || !isConnectedMediaElement(audio) || !hasBackgroundAudio) return;
    audio.playbackRate = reelSettings.playbackRate;
    audio.loop = true;
    const syncAudio = () => {
      if (!isConnectedMediaElement(audio) || !isConnectedMediaElement(video)) return;
      try {
        if (Math.abs(audio.currentTime - video.currentTime) > 0.18) audio.currentTime = video.currentTime;
      } catch {}
    };
    video.addEventListener("timeupdate", syncAudio);
    return () => video.removeEventListener("timeupdate", syncAudio);
  }, [hasBackgroundAudio, reelSettings.audioUrl, reelSettings.playbackRate]);

  const isYT =
    reelVideoUrl?.includes("youtube.com") ||
    reelVideoUrl?.includes("youtu.be");

  return (
    <>
    <div
      className="bg-black relative w-full feed-reel"
      style={{ aspectRatio: "9/16", height: "100svh", minHeight: "100svh", maxHeight: "100svh", touchAction: "pan-y" }}
    >
      {isYT ? (
        <div className="w-full h-full flex items-center justify-center bg-zinc-900">
          <Play size={40} className="text-white/40" />
        </div>
      ) : (
        <>
          <video
            ref={ref}
            className="w-full h-full object-cover"
            style={{ touchAction: "pan-y", filter: reelSettings.cssFilter }}
            loop
            muted
            playsInline
            crossOrigin="anonymous"
            preload="auto"
          >
            {!hlsPlayback.isHls && reelVideoUrl && (
              <source
                src={reelVideoUrl}
                type={getVideoMimeType(reelVideoUrl)}
              />
            )}
          </video>
          {isOwner && bunnyVideoId && (
            <>
              <BunnyCreatorProcessingOverlay state={bunnyVideoStatus} />
              {bunnyVideoStatus.phase === "failed" && (
                <BunnyVideoStatusOverlay
                  state={bunnyVideoStatus}
                  onRetry={bunnyVideoStatus.retry}
                />
              )}
            </>
          )}
          {hasBackgroundAudio && (
            <audio ref={audioRef} src={reelSettings.audioUrl || undefined} preload="metadata" loop muted />
          )}
        </>
      )}
      <div className="absolute inset-0 bg-gradient-to-t from-black/60 via-transparent to-transparent pointer-events-none" />
      {/* The media surface opens the immersive viewer; overlay controls remain
          independently clickable above it. */}
      <button
        type="button"
        aria-label="Open video in full-screen Reels"
        onClick={(event) => {
          event.stopPropagation();
          openFullReelsAt(String(post.id));
        }}
        className="absolute inset-0 z-10"
      />
      <button
        type="button"
        aria-label="Open in full-screen Reels"
        onClick={(event) => {
          event.stopPropagation();
          openFullReelsAt(String(post.id));
        }}
        className="absolute left-4 top-4 z-20 rounded-full border border-white/15 bg-black/70 px-3 py-2 text-xs font-bold text-white backdrop-blur-sm"
      >
        â†— Reels
      </button>
      <button
        type="button"
        aria-label="Reel actions"
        aria-expanded={showReelMenu}
        data-fame-reel-menu-trigger
        onPointerDown={toggleReelMenu}
        onKeyDown={toggleReelMenuFromKeyboard}
        className="absolute right-16 top-4 z-40 pointer-events-auto touch-manipulation grid h-9 w-9 place-items-center rounded-full border border-white/15 bg-black/70 text-white backdrop-blur-sm transition hover:bg-black/90 cursor-pointer"
        style={{ pointerEvents: "auto", touchAction: "manipulation" }}
      >
        <MoreVertical size={18} />
      </button>
      {showReelMenu &&
        createPortal(
          <div
            role="menu"
            data-fame-reel-menu
            onClick={(event) => event.stopPropagation()}
            onPointerDown={(event) => event.stopPropagation()}
            className="fixed z-[9999] min-w-44 max-w-[calc(100vw-24px)] max-h-[calc(100dvh-24px)] overflow-y-auto overscroll-contain rounded-xl border border-white/10 bg-[#160d12]/95 py-1 shadow-2xl backdrop-blur-xl pointer-events-auto"
            style={{
              top: reelMenuPosition.top,
              right: reelMenuPosition.right,
              pointerEvents: "auto",
            }}
          >
            {isOwner ? (
              <>
                <button
                  type="button"
                  role="menuitem"
                  onPointerDown={(event) =>
                    runReelMenuAction(event, "Edit Reel", onEdit)
                  }
                  onKeyDown={(event) =>
                    runReelMenuActionFromKeyboard(event, "Edit Reel", onEdit)
                  }
                  className="pointer-events-auto flex w-full items-center gap-2.5 px-3.5 py-3 text-left text-xs font-bold text-white/80 transition hover:bg-white/10 cursor-pointer"
                >
                  <SlidersHorizontal size={15} className="text-cyan-200" /> Edit Reel
                </button>
                <button
                  type="button"
                  role="menuitem"
                  onPointerDown={(event) =>
                    runReelMenuAction(event, "Delete Reel", () =>
                      setShowDeleteConfirm(true),
                    )
                  }
                  onKeyDown={(event) =>
                    runReelMenuActionFromKeyboard(event, "Delete Reel", () =>
                      setShowDeleteConfirm(true),
                    )
                  }
                  className="pointer-events-auto flex w-full items-center gap-2.5 border-t border-white/[0.07] px-3.5 py-3 text-left text-xs font-bold text-rose-300 transition hover:bg-rose-500/10 cursor-pointer"
                >
                  <Trash2 size={15} /> Delete Reel
                </button>
              </>
            ) : (
              <button
                type="button"
                role="menuitem"
                onPointerDown={(event) =>
                  runReelMenuAction(event, "Report Reel", onReport)
                }
                onKeyDown={(event) =>
                  runReelMenuActionFromKeyboard(event, "Report Reel", onReport)
                }
                className="pointer-events-auto flex w-full items-center gap-2.5 px-3.5 py-3 text-left text-xs font-bold text-orange-200 transition hover:bg-orange-500/10 cursor-pointer"
              >
                <Flag size={15} /> Report Reel
              </button>
            )}
          </div>,
          document.body,
        )}
      {showDeleteConfirm && (
        <div
          className="absolute inset-0 z-[100] flex items-center justify-center bg-black/75 p-5 backdrop-blur-sm"
          onClick={(event) => { event.preventDefault(); event.stopPropagation(); }}
        >
          <div
            role="alertdialog"
            aria-modal="true"
            aria-labelledby={`delete-reel-title-${post.id}`}
            className="w-full max-w-sm rounded-2xl border border-white/10 bg-zinc-950 p-5 shadow-2xl"
            onClick={(event) => event.stopPropagation()}
          >
            <h2 id={`delete-reel-title-${post.id}`} className="text-base font-black text-white">
              Delete this Reel?
            </h2>
            <p className="mt-2 text-sm leading-relaxed text-white/60">
              This permanently removes the Reel from your feed.
            </p>
            <div className="mt-5 flex gap-2">
              <button
                type="button"
                onClick={(event) => {
                  event.preventDefault();
                  event.stopPropagation();
                  setShowDeleteConfirm(false);
                }}
                className="flex-1 rounded-xl border border-white/10 px-4 py-3 text-sm font-bold text-white/75 hover:bg-white/5"
              >
                Cancel
              </button>
              <button
                type="button"
                onClick={(event) => {
                  event.preventDefault();
                  event.stopPropagation();
                  setShowDeleteConfirm(false);
                  onDelete();
                }}
                className="flex-1 rounded-xl bg-rose-600 px-4 py-3 text-sm font-black text-white hover:bg-rose-500"
              >
                Delete Reel
              </button>
            </div>
          </div>
        </div>
      )}
      {!isYT && (
        <button
          type="button"
          aria-label={audioNeedsGesture ? "Play selected background audio" : muted ? "Unmute video" : "Mute video"}
          title={audioNeedsGesture ? "Tap to play selected background audio" : undefined}
          onClick={() => {
            if (hasBackgroundAudio && audioNeedsGesture && !muted && audioRef.current) {
              const audio = audioRef.current;
              audio.muted = false;
              audio.volume = 1;
              void audio.play().then(() => setAudioNeedsGesture(false)).catch(() => setAudioNeedsGesture(true));
              return;
            }
            const nextMuted = !muted;
            setMuted(nextMuted);
            if (ref.current) ref.current.muted = hasBackgroundAudio || nextMuted;
            if (audioRef.current) audioRef.current.muted = nextMuted;
            if (hasBackgroundAudio && !nextMuted && audioRef.current) {
              const audio = audioRef.current;
              audio.volume = 1;
              void audio.play().then(() => setAudioNeedsGesture(false)).catch(() => setAudioNeedsGesture(true));
            }
          }}
          className="absolute top-4 right-4 z-20 p-2 bg-black/75 rounded-full border border-white/10"
        >
          {muted || audioNeedsGesture ? (
            <VolumeX size={16} className="text-white" />
          ) : (
            <Volume2 size={16} className="text-white" />
          )}
        </button>
      )}
      <div className="absolute bottom-5 left-4 right-16 z-20 max-h-[35%] overflow-y-auto overscroll-contain rounded-lg bg-black/75 p-3 text-white">
        <div className="mb-1.5 flex items-center gap-2">
          <ActiveStatusAvatar
            src={post.author_profile?.avatar_url || post.author_avatar}
            name={post.author_profile?.full_name || post.author}
            size={28}
            online={onlineUserIds.has(post.author_id || "")}
          />
          <p className="font-bold text-sm drop-shadow-lg">
          @{post.author_profile?.full_name || post.author || "user"}
          </p>
        </div>
        {post.content && (
          <p className="mt-1 whitespace-pre-wrap break-words text-xs leading-relaxed text-white/90">
            {maskProfanity(post.content)}
          </p>
        )}
      </div>
      <div className="reel-side-actions absolute right-3 bottom-16 z-20 flex flex-col items-center gap-4">
        <button
          onClick={toggleReelLike}
          disabled={likePending}
          aria-pressed={likedByMe}
          aria-label={likedByMe ? "Unlike Reel" : "Like Reel"}
          className="flex flex-col items-center disabled:opacity-70"
        >
          <Heart
            size={26}
            fill={likedByMe ? "#ff2d55" : "none"}
            className={likedByMe ? "text-[#ff2d55]" : "text-white"}
          />
          <span className="text-white text-[10px] font-bold mt-1">
            {likeCount}
          </span>
        </button>
        <button
          type="button"
          onClick={onOpenComments}
          aria-label={`Open ${commentCount} comments`}
          className="flex flex-col items-center"
        >
          <MessageCircle size={24} className="text-white" />
          <span className="mt-1 text-[10px] font-bold text-white">{commentCount}</span>
        </button>
        <button
          type="button"
          aria-label="Share Reel"
          onClick={(event) => {
            event.stopPropagation();
            const reelId = String(post.id);
            const authorName = post.author || post.author_name || "Flicks India";
            const caption = post.content || post.meta_description || "";
            // Always open the share popup modal (WhatsApp / Facebook / Instagram / native).
            // The modal's handleShare routes through shareToSocial which handles
            // Android bridge, navigator.share, and direct web-intent fallbacks.
            requestShareModal(
              {
                id: reelId,
                title: post.meta_title || post.content?.slice(0, 72) || "Flicks Reel",
                content: caption,
                media_url: post.media_url,
                video_url: post.video_url,
                cover_url: post.cover_url,
                type: post.type || "reel",
                author: authorName,
                share_url: buildPostShareUrl(reelId),
              },
              event.currentTarget,
            );
          }}
          className="flex flex-col items-center text-white"
        >
          <Share2 size={23} />
          <span className="mt-1 text-[10px] font-bold">Share</span>
        </button>
        <SavePostButton postId={String(post.id)} />
        <div className="reel-side-magnet">
          <MagnetButton postId={post.id} postType="post" postOwnerId={post.author_id || ""} currentUserId={currentUserId} dark label="Magnet" />
        </div>
      </div>
    </div>
    <section className="reel-engagement-footer" aria-label="Reel engagement">
      <div className="flex min-w-0 items-center gap-2">
        <div className="flex -space-x-2" aria-hidden="true">
          {viewerPreviews.length ? (
